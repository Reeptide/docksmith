# Docksmith — manual walkthrough
#
# `sudo make demo` runs all of this automatically with assertions.
# These are the same commands to type by hand, e.g. when demonstrating live.
# State lives in ~/.docksmith (resolved from SUDO_USER, not $HOME, under sudo).

# ── Setup ────────────────────────────────────────────────────────────────────
sudo make clean
make build
sudo ./setup/import-base-images.sh

# ── Build: cold, warm, and partial invalidation ──────────────────────────────
sudo ./docksmith build -t myapp:latest ./sample-app        # all CACHE MISS
sudo ./docksmith build -t myapp:latest ./sample-app        # all CACHE HIT, same digest

sudo sh -c 'echo "# modified" >> sample-app/config/settings.txt'
sudo ./docksmith build -t myapp:latest ./sample-app        # that COPY misses, cascade follows

sudo ./docksmith images
sudo cat ~/.docksmith/images/myapp_latest.json | python3 -m json.tool
sudo ls -lh ~/.docksmith/layers/
sudo cat ~/.docksmith/cache/index.json | python3 -m json.tool

# ── Run ──────────────────────────────────────────────────────────────────────
sudo ./docksmith run myapp:latest
sudo ./docksmith run -e GREETING=Howdy myapp:latest
sudo ./docksmith run myapp:latest /bin/sh -c 'echo custom command'

# Isolation: this file must NOT appear on the host
sudo ./docksmith run myapp:latest /bin/sh -c 'echo SECRET > /tmp/escape.txt'
ls /tmp/escape.txt                                          # expect: no such file

# pivot_root left nothing behind, and no mounts leaked
sudo ./docksmith run myapp:latest /bin/sh -c 'ls -a / | grep oldroot'   # expect: nothing
grep docksmith /proc/mounts                                            # expect: nothing

# ── Whiteouts: deletions survive layer reassembly ────────────────────────────
mkdir -p /tmp/wh && printf 'FROM busybox:latest\nRUN rm /bin/ls\nCMD ["/bin/true"]\n' > /tmp/wh/Docksmithfile
sudo ./docksmith build -t wh:1 /tmp/wh
sudo ./docksmith run wh:1 /bin/sh -c 'ls /bin/ls'          # expect: not found
sudo ./docksmith run busybox:latest /bin/ls /bin/ls        # base image undamaged

# ── Volumes ──────────────────────────────────────────────────────────────────
sudo ./docksmith run --rm -v /tmp/dsvol:/data myapp:latest /bin/sh -c 'echo hi > /data/f.txt'
cat /tmp/dsvol/f.txt
sudo ./docksmith run --rm -v /tmp/dsvol:/data:ro myapp:latest /bin/sh -c 'echo x > /data/no.txt'  # expect: fails

# ── Container lifecycle ──────────────────────────────────────────────────────
sudo ./docksmith run -d --name web myapp:latest /bin/sh -c 'echo up; sleep 300'
sudo ./docksmith ps
sudo ./docksmith logs web
sudo ./docksmith stop web        # near-instant: the init shim handles SIGTERM
sudo ./docksmith ps -a
sudo ./docksmith rm web

# ── Networking ───────────────────────────────────────────────────────────────
ip link show docksmith0
sudo ./docksmith run --rm myapp:latest /bin/sh -c 'ip addr show eth0'
sudo ./docksmith run --rm myapp:latest /bin/sh -c 'ping -c1 172.30.0.1'
sudo ./docksmith run --rm myapp:latest /bin/sh -c 'ping -c1 8.8.8.8'          # NAT
sudo ./docksmith run --rm myapp:latest /bin/sh -c 'nslookup example.com'      # DNS
sudo ./docksmith run --rm --net=none myapp:latest /bin/sh -c 'ping -c1 -W1 172.30.0.1'  # expect: fails

# Published ports, reachable from both localhost and the LAN address
sudo ./docksmith run -d --name srv -p 18080:8080 myapp:latest /bin/sh -c \
  'while true; do echo -e "HTTP/1.1 200 OK\r\n\r\nhello" | nc -l -p 8080; done'
curl -s http://127.0.0.1:18080
sudo iptables -t nat -S | grep 18080
sudo ./docksmith rm -f srv
sudo iptables -t nat -S | grep 18080          # expect: nothing, rules torn down

# Builds are offline
mkdir -p /tmp/nc && printf 'FROM busybox:latest\nRUN ping -c1 -W1 8.8.8.8 && echo ON > /n || echo OFF > /n\nCMD ["/bin/cat","/n"]\n' > /tmp/nc/Docksmithfile
sudo ./docksmith build --no-cache -t nc:1 /tmp/nc
sudo ./docksmith run --rm nc:1                 # expect: OFF

# ── save / load ──────────────────────────────────────────────────────────────
sudo ./docksmith save -o /tmp/myapp.tar myapp:latest
tar tf /tmp/myapp.tar
sudo DOCKSMITH_ROOT=/tmp/otherstore ./docksmith load -i /tmp/myapp.tar
sudo DOCKSMITH_ROOT=/tmp/otherstore ./docksmith images     # same digest as the original

# Corruption is detected, and nothing is written
python3 -c "d=bytearray(open('/tmp/myapp.tar','rb').read()); d[len(d)//2]^=0xff; open('/tmp/bad.tar','wb').write(d)"
sudo DOCKSMITH_ROOT=/tmp/badstore ./docksmith load -i /tmp/bad.tar   # expect: "is corrupt"

# ── rmi reference counting and prune ─────────────────────────────────────────
sudo ./docksmith rmi wh:1
sudo ./docksmith run --rm myapp:latest /bin/true    # sibling still works: shared base kept
sudo ./docksmith prune                              # prompts before removing
sudo ./docksmith prune -f

# ── Tests ────────────────────────────────────────────────────────────────────
make test
sudo make demo
